CREATE PROCEDURE sps_login_usuario
(
    @i_us_login VARCHAR(20)='bory', 
    @i_us_password VARCHAR(50) ='123'
)
AS
BEGIN
    SET NOCOUNT ON; -- Buena práctica para evitar mensajes de "n filas afectadas" extra
    
    DECLARE @w_numero_login INT;

    -- PASO 1: VERIFICAR SI EXISTE EL USUARIO
    IF EXISTS (SELECT 1,1 FROM ap_usuario WHERE us_login = @i_us_login)
    BEGIN 
        -- PASO 2: VERIFICAR SI ESTÁ ACTIVO (NO BLOQUEADO)
		IF EXISTS( SELECT 1, 1 FROM ap_usuario WHERE us_login = @i_us_login AND us_activo = 1 )
        BEGIN 
            -- PASO 3: VERIFICAR CONTRASEÑA
            IF EXISTS (SELECT us_id  FROM ap_usuario WHERE us_login = @i_us_login AND us_password = @i_us_password)
            BEGIN 
                -- A) LOGIN EXITOSO
				 -- Reiniciar contador de intentos a 0
                UPDATE ap_usuario SET us_numero_login = 0, us_fecha_ultimo_ingreso = GETDATE()
                WHERE us_login = @i_us_login
                SELECT
                    us_id AS 'IdUsuario', 
                    us_nombre AS 'Nombre', 
                    us_apellido AS 'Apellido', 
                    us_login AS 'Usuario', 
		    us_idRol AS 'IdRol',
                    us_password AS 'Password', 
                    us_telefono AS 'Telefono', 
                    us_cedula AS 'Cedula', 
                    us_fecha_nacimiento AS 'FechaNacimiento', 
                    us_fecha_registro AS 'FechaRegistro', 
                    us_fecha_actualizacion AS 'FechaActualizacion',
                    us_numero_login AS 'NumeroLogin', 
                    us_activo AS 'Activo'
                FROM ap_usuario
                WHERE us_login = @i_us_login AND us_password = @i_us_password

               
            END
            ELSE
            BEGIN 
                -- B) CONTRASEÑA INCORRECTA
                -- Aumentamos intentos
                UPDATE ap_usuario SET us_numero_login = us_numero_login + 1
                WHERE us_login = @i_us_login

                -- Verificamos si llegamos al límite (3)
                SELECT @w_numero_login = us_numero_login 
                FROM ap_usuario
                WHERE us_login = @i_us_login

                IF(@w_numero_login >= 3) 
                BEGIN
                    -- Bloquear usuario
                    UPDATE ap_usuario SET us_activo = 0
                    WHERE us_login = @i_us_login
                    
                    PRINT 'RETURN 1002' -- CÓDIGO: Usuario bloqueado tras 3 intentos
                END
                ELSE
                BEGIN
                    PRINT 'RETURN 1001' -- CÓDIGO: Contraseña incorrecta
                END
            END 
        END
        ELSE
        BEGIN
             PRINT 'RETURN 1002' -- CÓDIGO: Usuario Bloqueado / Inactivo (Existe pero activo=0)
        END
    END
    ELSE 
    BEGIN
        PRINT 'RETURN 1004' -- CÓDIGO: No existe usuario 
    END
END
GO

ALTER PROCEDURE [dbo].[sps_consultar_abonos_cliente]
    @i_idCliente INT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        p.pg_id AS idPago,
        p.pg_fecha_registro AS fecha,
        p.pg_abono AS abonoRealizado,
        -- Calculamos el saldo directamente de los valores registrados
        (p.pg_montoTotal - p.pg_abono) AS saldo, 
        p.pg_nombreUsuario AS usuario
    FROM dbo.ap_pagos p
    -- La clave es unir el pago (ID 4) con cualquiera de los ensayos del cliente 1 que tengan ese mismo ID de prueba
    WHERE p.pg_idEnsayo IN (
        SELECT DISTINCT en_idPrueba 
        FROM dbo.ap_ensayo 
        WHERE en_idCliente = @i_idCliente
    )
    ORDER BY p.pg_id DESC;
END

CREATE PROCEDURE [dbo].[sps_obtener_clientes_deudores]
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        CONCAT(c.cl_nombre, ' ', c.cl_apellido) AS NombreCompleto,
        e.en_nombreUsuario AS Ensayo,
        SUM(p.pg_abono) AS TotalAbonado,
        MAX(p.pg_montoTotal) AS TotalAPagar,
        (MAX(p.pg_montoTotal) - SUM(p.pg_abono)) AS SaldoPendiente,
        MAX(p.pg_fecha_registro) AS FechaUltimoMovimiento
    FROM dbo.ap_cliente c
    INNER JOIN dbo.ap_ensayo e ON c.cl_id = e.en_idCliente
    INNER JOIN dbo.ap_pagos p ON e.en_idPrueba = p.pg_idEnsayo
    GROUP BY c.cl_id, c.cl_nombre, c.cl_apellido, e.en_nombreUsuario
    HAVING SUM(p.pg_abono) < MAX(p.pg_montoTotal);
END
GO

    FROM dbo.ap_cliente c
    INNER JOIN dbo.ap_ensayo e ON c.cl_id = e.en_idCliente
    LEFT JOIN dbo.ap_pagos p ON e.en_idPrueba = p.pg_idEnsayo
    GROUP BY c.cl_id, c.cl_nombre, e.en_nombreUsuario
    HAVING (MAX(e.en_monto) - ISNULL(SUM(p.pg_abono), 0)) > 0;
END
GO
